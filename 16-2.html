一、对象（Object）

1. 创建对象

// 方式一：字面量（最常用） var cmq_person = { name: "小美", age: 20, hobby: "画画" }; // 方式二：new Object() var cmq_o = new Object(); cmq_o.name = "小美";

2. 读 & 写属性

// 点语法 cmq_person.name // "小美" cmq_person.age = 21; // 修改 // 中括号语法（属性名是变量/特殊字符时） var k = "hobby"; cmq_person[k] // "画画" cmq_person["city"] = "北京"; // 动态添加

3. 删除属性 delete 修改

delete cmq_person.hobby; // 彻底移除

4. Object.keys / values / entries 不修改

Object.keys(cmq_person); // ["name","age","hobby"] Object.values(cmq_person); // ["小美",20,"画画"] Object.entries(cmq_person); // [["name","小美"],["age",20],...]
等待运行…

5. 合并 & 展开 & 解构

// Object.assign 合并 var cmq_p2 = Object.assign({}, cmq_person, { age: 22, city: "上海" }); // 展开运算符（更现代） var cmq_p3 = { ...cmq_person, age: 22 }; // 解构赋值 var { name, age } = cmq_person;

二、字符串（String）⚠️ 不可变！所有方法返回新串

1. length & 索引访问

var cmq_s = "Hello"; cmq_s.length // 5 cmq_s[0] // "H" cmq_s.charAt(4) // "o"
等待运行…

2. 查找：indexOf / includes / startsWith / endsWith

cmq_s.indexOf("lo") // 3 找到的位置，否则 -1 cmq_s.includes("Java") // true / false cmq_s.startsWith("He") // true cmq_s.endsWith("!") // true
等待运行…

3. 截取：slice / substring / substr

// slice(起, 止) 含头不含尾，支持负数 cmq_s.slice(0, 5) // "Hello" cmq_s.slice(-6) // "JavaScript!" 倒数第6个到末尾 // substring(起, 止) 不支持负数，小的自动当起点 // substr(起, 长度) 老方法，尽量用 slice
等待运行…

4. 大小写 & 空白

cmq_s.toUpperCase() // 全大写 cmq_s.toLowerCase() // 全小写 " Hi ".trim() // "Hi" 去两端空格
等待运行…

5. replace / split / concat

// replace(找什么, 换成什么) cmq_s.replace("JavaScript", "JS") // split(分隔符) → 数组 "a,b,c".split(",") // ["a","b","c"] // concat 拼接（不如用 +） "Hi " + "小美" // "Hi 小美"
等待运行…

6. 模板字符串（反引号 `）✨

var name = "小美", age = 20; var s = `我叫 ${name}，今年 ${age} 岁。`; // 比 "我叫 " + name + "，今年 " + age + " 岁。" 清爽多了
